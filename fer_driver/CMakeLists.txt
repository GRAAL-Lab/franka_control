cmake_minimum_required(VERSION 3.16)
project(fer_driver)
find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
option(FER_BUILD_HARDWARE "Build the libfranka hardware executables" ON)
if(FER_BUILD_HARDWARE)
  find_package(Franka 0.9 REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(fer_msgs REQUIRED)
  find_package(fer_model REQUIRED)
  find_package(sensor_msgs REQUIRED)
  foreach(node driver_node gripper_node)
    add_executable(${node} src/${node}.cpp)
    target_compile_features(${node} PRIVATE cxx_std_17)
    target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
    target_include_directories(${node} PRIVATE include)
    target_link_libraries(${node} Franka::Franka fer_model::fer_model Threads::Threads)
    ament_target_dependencies(${node} rclcpp fer_msgs sensor_msgs)
    install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
  endforeach()
endif()
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY conf launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_latest_value test/test_latest_value.cpp)
  target_compile_features(test_latest_value PRIVATE cxx_std_17)
  target_include_directories(test_latest_value PRIVATE include)
  target_link_libraries(test_latest_value Threads::Threads)
endif()
ament_package()
