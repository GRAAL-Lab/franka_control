cmake_minimum_required(VERSION 3.16)
project(fer_ctrl)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(fer_msgs REQUIRED)
find_package(fer_model REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_library(fer_controller src/controller.cpp)
target_compile_features(fer_controller PUBLIC cxx_std_17)
target_compile_options(fer_controller PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(fer_controller PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fer_controller PUBLIC fer_model::fer_model)
add_executable(kinematic_control_node src/kinematic_control_node.cpp)
target_link_libraries(kinematic_control_node fer_controller)
ament_target_dependencies(kinematic_control_node rclcpp fer_msgs geometry_msgs tf2_ros)
install(TARGETS fer_controller EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS kinematic_control_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY conf launch DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(fer_model)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_controller test/test_controller.cpp)
  target_link_libraries(test_controller fer_controller)
endif()
ament_package()
