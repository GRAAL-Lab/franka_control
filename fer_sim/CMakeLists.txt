cmake_minimum_required(VERSION 3.16)
project(fer_sim)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(fer_msgs REQUIRED)
find_package(fer_model REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
foreach(node simulator_node visualizer_node)
  add_executable(${node} src/${node}.cpp)
  target_compile_features(${node} PRIVATE cxx_std_17)
  target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(${node} fer_model::fer_model)
  ament_target_dependencies(${node} rclcpp fer_msgs sensor_msgs visualization_msgs)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY conf launch DESTINATION share/${PROJECT_NAME})
ament_package()
