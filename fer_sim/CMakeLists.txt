cmake_minimum_required(VERSION 3.16)
project(fer_sim)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(fer_msgs REQUIRED)
find_package(fer_model REQUIRED)
find_package(fer_ctrl REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)

add_executable(simulator_node src/simulator_node.cpp src/fer_sim/simulator.cpp)
add_executable(visualizer_node src/visualizer_node.cpp src/fer_sim/robot_visualizer.cpp)
foreach(target simulator_node visualizer_node)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_include_directories(${target} PRIVATE include)
    target_link_libraries(${target} fer_ctrl::fer_dynamic_ctrl_lib)
    ament_target_dependencies(${target} rclcpp fer_msgs sensor_msgs visualization_msgs ament_index_cpp)
    install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endforeach()

install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY conf launch DESTINATION share/${PROJECT_NAME})
ament_package()
